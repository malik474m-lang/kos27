# Кодировка
AddDefaultCharset UTF-8

# Перенаправление всех запросов на index.php (Front Controller)
RewriteEngine On

# Исключения: реальные файлы и директории
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d

# Всё остальное → index.php
RewriteRule ^(.*)$ index.php [QSA,L]

# Кэширование статики
<IfModule mod_expires.c>
    ExpiresActive On
    ExpiresByType image/jpeg "access plus 1 month"
    ExpiresByType image/png "access plus 1 month"
    ExpiresByType image/svg+xml "access plus 1 month"
    ExpiresByType image/webp "access plus 1 month"
    ExpiresByType text/css "access plus 1 week"
    ExpiresByType application/javascript "access plus 1 week"
    ExpiresByType image/x-icon "access plus 1 year"
    ExpiresByType application/manifest+json "access plus 1 week"
</IfModule>

# Сжатие
<IfModule mod_deflate.c>
    AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json image/svg+xml application/manifest+json
</IfModule>

# MIME типы для PWA
<IfModule mod_mime.c>
    AddType application/manifest+json .webmanifest .json
    AddType application/javascript .js
</IfModule>

# Безопасность - блокируем .env и логи, но НЕ manifest.json и service-worker.js
<FilesMatch "\.(env|log)$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
</FilesMatch>

# Разрешаем manifest.json (для PWA)
<Files "manifest.json">
    <IfModule mod_authz_core.c>
        Require all granted
    </IfModule>
</Files>

# Разрешаем service-worker.js (для PWA)
<Files "service-worker.js">
    <IfModule mod_authz_core.c>
        Require all granted
    </IfModule>
    # Service Worker не должен кэшироваться
    <IfModule mod_headers.c>
        Header set Cache-Control "no-cache, no-store, must-revalidate"
    </IfModule>
</Files>

# Разрешаем offline.html
<Files "offline.html">
    <IfModule mod_authz_core.c>
        Require all granted
    </IfModule>
</Files>

# Разрешаем site-settings.json для PHP (но не напрямую)
<Files "site-settings.json">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
</Files>

# Запрет доступа к служебным файлам
<FilesMatch "^\.">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
</FilesMatch>
